3) A la suite de EscrowVault (n'importe ou dans le fichier, hors d'un modele), coller :

enum PayoutStatus {
  TO_SEND
  SENDING
  PROCESSING
  PAID
  FAILED
}

model Payout {
  id               String       @id @default(uuid())
  escrowId         String       @unique
  escrow           EscrowVault  @relation(fields: [escrowId], references: [id], onDelete: Cascade)
  hostUserId       String?
  amount           Decimal      @db.Decimal(10, 2)
  status           PayoutStatus @default(TO_SEND)
  attempts         Int          @default(0)
  reference        String?      @unique
  gatewayRef       String?
  beneficiaryChannel String?
  beneficiaryPhone   String?
  beneficiaryName    String?
  failureReason    String?
  sentById         String?
  sentAt           DateTime?
  paidAt           DateTime?
  createdAt        DateTime     @default(now())
  updatedAt        DateTime     @updatedAt

  @@index([status, createdAt])
}
